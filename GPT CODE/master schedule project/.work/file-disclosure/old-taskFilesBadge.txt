function taskFilesBadge(task){const catalog=task.fileCatalog,count=Array.isArray(catalog?.files)?Math.min(catalog.files.length,FILE_LIMIT):0,selected=selectedCatalogFiles(catalog);return '<details class="task-files-disclosure" data-files-task="'+escapeHTML(task.id)+'" '+(expandedFileLists.has(task.id)?'open':'')+'><summary title="ยุบหรือขยายไฟล์ประกอบงาน '+escapeHTML(task.name)+'"><span class="file-disclosure-arrow" aria-hidden="true"></span><span class="file-disclosure-label">ไฟล์ที่เลือก '+selected.length.toLocaleString('th-TH')+' / '+count.toLocaleString('th-TH')+'</span></summary><div class="task-files-body">'+(selected.length?'<ul class="task-file-preview">'+selected.slice(0,5).map(f=>'<li title="'+escapeHTML(f.path)+'">'+fileLinkMarkup(catalog,f)+(fileParent(f.path)?'<small>'+escapeHTML(fileParent(f.path))+'</small>':'')+'</li>').join('')+'</ul>':'<p class="task-files-empty">ยังไม่มีไฟล์ที่เลือก</p>')+'<button type="button" class="task-selected-more" data-task-files="'+escapeHTML(task.id)+'" data-selected-only="'+(selected.length>5)+'">'+(selected.length>5?'ดูที่เลือกทั้งหมด ('+selected.length.toLocaleString('th-TH')+')':'ดูรายการไฟล์ทั้งหมด')+'</button></div></details>'}
